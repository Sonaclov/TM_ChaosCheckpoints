/**
 *	Chaos Checkpoints
 *	----------------
 *	A fast, free-roam Trackmania 2020 game mode.
 *
 *	- Players spawn at RANDOM landmarks of the map (starts / checkpoints).
 *	- A set of "chaos checkpoints" (virtual, proximity-based) is active on
 *	  the map at all times. Drive into one to claim it and score points.
 *	- A claimed checkpoint instantly respawns somewhere else on the map.
 *	- Some checkpoints are GOLDEN and worth 3 points instead of 1.
 *	- First player to reach the points limit wins the round instantly.
 *	  Otherwise the leader when the round timer runs out wins.
 *	- Rounds are short by design: the time limit is clamped to 60-600s
 *	  (default 480s = 8 minutes, hard max 10 minutes).
 *	- The player with the most points after all rounds wins the map.
 *
 *	Optional extras, all controlled by server settings:
 *	- Combos: chain claims within a time window for bonus points.
 *	- Frenzy finale: during the last seconds every checkpoint is golden
 *	  and one extra checkpoint is active.
 *	- Shuffle: all unclaimed checkpoints periodically teleport.
 *	- Overtime: on a tied round, sudden death — first claim wins.
 *	- Golden steal: golden checkpoints also steal a point from the leader.
 *	- First-claim bonus: extra points for the first claim of the round.
 *	- HUD distance: live distance to the nearest checkpoint.
 *
 *	The chaos checkpoints are fully virtual: they reuse the positions of the
 *	map's landmarks (checkpoints, starts, finish) as candidate locations, but
 *	claiming is done by distance check, not by the map's race route. Any
 *	regular TM_Race map with a decent number of checkpoints works.
 */

#Const C_PointsGolden 3				// Points for a golden chaos checkpoint
#Const C_ComboMaxBonus 3			// Max bonus points a combo chain can add per claim
#Const C_OvertimeMaxMs 60000		// Hard cap on overtime length
#Const C_SpawnCountdownMs 3000		// Countdown before players are released at round start
#Const C_RespawnDelayMs 1500		// Delay before an unspawned player is thrown back in
#Const C_EndRoundPauseMs 7000		// Pause between rounds to show the results

#Const C_MarkerHeightOffset <0., 3., 0.>

#Struct K_ChaosCP {

// ---------------------------------- //
// Small helpers around tunable settings
// ---------------------------------- //
Integer GetRoundTimeLimitMs() {
	// Fast rounds by design: clamp between 1 and 10 minutes
	return ML::Clamp(S_RoundTimeLimit, 60, 600) * 1000;
}

Integer GetNbChaosCheckpoints() {
	// Frenzy finale runs one extra checkpoint
	declare Nb = ML::Clamp(S_NbChaosCheckpoints, 1, 8);
	if (G_FrenzyActive) Nb += 1;
	return Nb;
}

// ---------------------------------- //
// Score helpers
//
// All interactions with the Scores library are centralized here so the mode
// is easy to adjust if the library API changes between script pack versions.
// ---------------------------------- //
Void ResetRoundPoints() {
	foreach (Score in Scores) {
		Scores::SetPlayerRoundPoints(Score, 0);
	}
}

Void AddRoundPoints(CSmScore _Score, Integer _Points) {
	if (_Score == Null) return;
	Scores::AddPlayerRoundPoints(_Score, _Points);
}

Void SetRoundPoints(CSmScore _Score, Integer _Points) {
	if (_Score == Null) return;
	Scores::SetPlayerRoundPoints(_Score, _Points);
}

Integer GetRoundPoints(CSmScore _Score) {
	if (_Score == Null) return 0;
	return Scores::GetPlayerRoundPoints(_Score);
}

Integer GetMatchPoints(CSmScore _Score) {
	if (_Score == Null) return 0;
	return Scores::GetPlayerMatchPoints(_Score);
}

// Move this round's points into the match total
Void TransferRoundPointsToMatch() {
	foreach (Score in Scores) {
		Scores::AddPlayerMatchPoints(Score, Scores::GetPlayerRoundPoints(Score));
		Scores::SetPlayerRoundPoints(Score, 0);
	}
}

// ---------------------------------- //
/** Best score of the round (Null if nobody scored)
 */
CSmScore GetRoundLeader() {
	declare CSmScore Leader <=> Null;
	declare LeaderPoints = 0;
	foreach (Score in Scores) {
		declare Points = GetRoundPoints(Score);
		if (Points > LeaderPoints) {
			LeaderPoints = Points;
			Leader <=> Score;
		}
	}
	return Leader;
}

CSmScore GetMatchLeader() {
	declare CSmScore Leader <=> Null;
	declare LeaderPoints = 0;
	foreach (Score in Scores) {
		declare Points = GetMatchPoints(Score);
		if (Points > LeaderPoints) {
			LeaderPoints = Points;
			Leader <=> Score;
		}
	}
	return Leader;
}

// ---------------------------------- //
/** True when 2+ players share the top round score.
 *	Used to decide whether overtime should start.
 */
Boolean IsRoundTied() {
	if (Players.count < 2) return False;
	declare Best = -1;
	declare NbAtBest = 0;
	foreach (Score in Scores) {
		declare Points = GetRoundPoints(Score);
		if (Points > Best) {
			Best = Points;
			NbAtBest = 1;
		} else if (Points == Best) {
			NbAtBest += 1;
		}
	}
	return NbAtBest >= 2;
}

// ---------------------------------- //
/** Collect the landmarks usable as spawn points and checkpoint locations
 */
Void BuildCandidateLandmarks() {
	G_CandidateLandmarks.clear();
	foreach (Index => Landmark in MapLandmarks) {
		// Keep race waypoints (start, checkpoints, finish) and spawn blocks
		if (Landmark.Waypoint == Null && Landmark.Tag != "Spawn") continue;
		G_CandidateLandmarks.add(Index);
	}

	if (G_CandidateLandmarks.count < 4) {
		Chat("$f80Warning: this map has very few landmarks ($fff" ^ G_CandidateLandmarks.count ^ "$f80). Chaos Checkpoints plays best on maps with many checkpoints!");
	}
}

// ---------------------------------- //
/** True if a chaos checkpoint currently sits on this landmark
 */
Boolean IsLandmarkUsed(Integer _LandmarkIndex) {
	foreach (CP in G_ActiveCPs) {
		if (CP.LandmarkIndex == _LandmarkIndex) return True;
	}
	return False;
}

// ---------------------------------- //
/** Smallest distance from a position to any spawned player
 */
Real DistanceToClosestPlayer(Vec3 _Position) {
	declare Closest = 999999.;
	foreach (Player in Players) {
		if (Player.SpawnStatus != CSmPlayer::ESpawnStatus::Spawned) continue;
		declare Distance = ML::Distance(Player.Position, _Position);
		if (Distance < Closest) Closest = Distance;
	}
	return Closest;
}

// ---------------------------------- //
/** Spawn one new chaos checkpoint at a random free landmark.
 *	Prefers locations at least S_MinCheckpointDistance away from every
 *	player; falls back to the farthest candidate found.
 *	During the frenzy finale (or when _ForceGolden is set) the checkpoint
 *	is always golden.
 */
Void SpawnChaosCP(Integer _ExcludedLandmark, Boolean _ForceGolden) {
	if (G_CandidateLandmarks.count <= 0) return;

	declare BestIndex = -1;
	declare BestDistance = -1.;
	for (Try, 1, C_MaxPickTries) {
		declare LandmarkIndex = G_CandidateLandmarks[ML::Rand(0, G_CandidateLandmarks.count - 1)];
		if (LandmarkIndex == _ExcludedLandmark) continue;
		if (IsLandmarkUsed(LandmarkIndex)) continue;
		if (!MapLandmarks.existskey(LandmarkIndex)) continue;

		declare Distance = DistanceToClosestPlayer(MapLandmarks[LandmarkIndex].Position);
		if (Distance >= S_MinCheckpointDistance) {
			BestIndex = LandmarkIndex;
			break;
		}
		if (Distance > BestDistance) {
			BestDistance = Distance;
			BestIndex = LandmarkIndex;
		}
	}
	if (BestIndex < 0 || !MapLandmarks.existskey(BestIndex)) return;

	declare IsGolden = _ForceGolden || G_FrenzyActive || ML::Rand(1, 100) <= ML::Clamp(S_GoldenChance, 0, 100);
	declare CP = K_ChaosCP {
		LandmarkIndex = BestIndex,
		Position = MapLandmarks[BestIndex].Position,
		Points = C_PointsNormal,
		IsGolden = IsGolden
	};
	if (IsGolden) CP.Points = C_PointsGolden;
	G_ActiveCPs.add(CP);
}

// ---------------------------------- //
/** Rebuild the 3D markers on every active chaos checkpoint,
 *	and push the positions to the clients for the HUD distance display.
 */
Void RefreshMarkers() {
	UIManager.UIAll.ClearMarkers();
	foreach (CP in G_ActiveCPs) {
		declare Marker = UIManager.UIAll.AddMarker(CP.Position + C_MarkerHeightOffset);
		if (CP.IsGolden) {
			Marker.Label = "$fd0★ +" ^ CP.Points;
		} else {
			Marker.Label = "$0cf◆ +" ^ CP.Points;
		}
		Marker.HudVisibility = CUIConfigMarker::EHudVisibility::Always;
	}

	declare netwrite Vec3[] Net_ChaosCP_Positions for Teams[0];
	declare Vec3[] Positions;
	foreach (CP in G_ActiveCPs) {
		Positions.add(CP.Position);
	}
	Net_ChaosCP_Positions = Positions;
}

// ---------------------------------- //
// Custom HUD (top right): round timer, round/points info, standings and
// optional nearest-checkpoint distance. Data is pushed through netwrite
// variables; the layout itself is static.
// ---------------------------------- //
Void SetHudEndTime(Integer _EndTime) {
	declare netwrite Integer Net_ChaosCP_EndTime for Teams[0] = -1;
	Net_ChaosCP_EndTime = _EndTime;
}

Void RefreshHud() {
	declare netwrite Text Net_ChaosCP_Header for Teams[0] = "";
	declare netwrite Text Net_ChaosCP_Standings for Teams[0] = "";
	declare netwrite Boolean Net_ChaosCP_ShowDist for Teams[0] = False;

	Net_ChaosCP_ShowDist = S_HudShowDistance;

	declare Header = "Round " ^ G_RoundNb ^ "/" ^ ML::Max(S_RoundsPerMap, 1);
	if (S_PointsLimit > 0) Header ^= " · First to " ^ S_PointsLimit;
	if (G_InOvertime) Header = "OVERTIME — first claim wins!";
	else if (G_FrenzyActive) Header ^= " · $fd0FRENZY!";
	Net_ChaosCP_Header = Header;

	// Sort players by (round + match) points, descending
	declare Integer[Text] SortByLogin;
	declare Text[Text] NameByLogin;
	foreach (Score in Scores) {
		if (Score.User == Null) continue;
		SortByLogin[Score.User.Login] = -(GetRoundPoints(Score) + GetMatchPoints(Score));
		NameByLogin[Score.User.Login] = Score.User.Name;
	}
	SortByLogin = SortByLogin.sort();

	declare Standings = "";
	declare Rank = 0;
	foreach (Login => NegPoints in SortByLogin) {
		Rank += 1;
		if (Rank > 5) break;
		declare Name = Login;
		if (NameByLogin.existskey(Login)) Name = NameByLogin[Login];
		Standings ^= Rank ^ ". $<$fff" ^ Name ^ "$> $ff0" ^ (-NegPoints) ^ "\n";
	}
	Net_ChaosCP_Standings = Standings;
}

// ---------------------------------- //
/** Spawn (or respawn) a player, at a random landmark when enabled.
 *
 *	Centralized here on purpose: this is the only place calling into the
 *	Race library's spawn functions.
 */
Void ChaosSpawnPlayer(CSmPlayer _Player, Integer _StartTime) {
	if (_Player == Null) return;

	if (S_RandomPlayerSpawn && G_CandidateLandmarks.count > 0) {
		// Try to pick a landmark that has no chaos checkpoint sitting on it,
		// so nobody spawns straight into free points
		declare PickedIndex = -1;
		for (Try, 1, C_MaxPickTries) {
			declare LandmarkIndex = G_CandidateLandmarks[ML::Rand(0, G_CandidateLandmarks.count - 1)];
			PickedIndex = LandmarkIndex;
			if (!IsLandmarkUsed(LandmarkIndex)) break;
		}
		if (PickedIndex >= 0 && MapLandmarks.existskey(PickedIndex)) {
			declare Landmark <=> MapLandmarks[PickedIndex];
			Race::Start(_Player, Landmark.Tag, Landmark.Order, _StartTime);
			return;
		}
	}

	// Fallback: regular spawn at the map start
	Race::Start(_Player, _StartTime);
}

// ---------------------------------- //
// Mode flow
// ---------------------------------- //

***Match_StartServer***
***
// Custom HUD layer, kept for the whole server lifetime
G_HudLayer <=> UIManager.UILayerCreate();
G_HudLayer.Type = CUILayer::EUILayerType::Normal;
G_HudLayer.ManialinkPage = GetHudManialink();
UIManager.UIAll.UILayers.add(G_HudLayer);
SetHudEndTime(-1);
***

***Match_StartMap***
***
G_RoundNb = 0;
G_ActiveCPs.clear();
BuildCandidateLandmarks();
Chat("$0cfChaos Checkpoints$fff — random spawns, grab the marked checkpoints, first to $ff0" ^ S_PointsLimit ^ "$fff points wins the round!");
***

***Match_StartRound***
***
G_RoundNb += 1;
G_RoundEndRequested = False;
G_FirstClaimDone = False;
G_FrenzyActive = False;
G_InOvertime = False;
G_ActiveCPs.clear();
G_RespawnAt.clear();
G_LastClaimTime.clear();
G_ComboCount.clear();
ResetRoundPoints();

// Everyone spawns somewhere random, with a short countdown
declare SpawnTime = Now + C_SpawnCountdownMs;
foreach (Player in Players) {
	ChaosSpawnPlayer(Player, SpawnTime);
}

// Round timer: default 8 minutes, never more than 10
G_RoundEndTime = SpawnTime + GetRoundTimeLimitMs();
SetHudEndTime(G_RoundEndTime);

// First checkpoint shuffle, if enabled
G_NextShuffleTime = -1;
if (S_ShuffleInterval > 0) {
	G_NextShuffleTime = SpawnTime + ML::Clamp(S_ShuffleInterval, 15, 600) * 1000;
}

// Initial batch of chaos checkpoints
for (I, 1, GetNbChaosCheckpoints()) {
	SpawnChaosCP(-1, False);
}
RefreshMarkers();
RefreshHud();

Announce("Round " ^ G_RoundNb ^ " — GO!");
Chat("Round $ff0" ^ G_RoundNb ^ "/" ^ ML::Max(S_RoundsPerMap, 1) ^ "$fff started — $ff0" ^ (GetRoundTimeLimitMs() / 60000) ^ "$fff minutes on the clock.");
***

***Match_PlayLoop***
***
// ---------------------------------- //
// 1. Throw unspawned players back in (join, finish, give up, crash out).
//    Giving up is a legit tactic: it rerolls your position.
foreach (Player in Players) {
	if (Player.SpawnStatus == CSmPlayer::ESpawnStatus::NotSpawned) {
		declare Login = Player.User.Login;
		if (!G_RespawnAt.existskey(Login)) {
			G_RespawnAt[Login] = Now + C_RespawnDelayMs;
		} else if (Now >= G_RespawnAt[Login]) {
			declare Removed = G_RespawnAt.removekey(Login);
			ChaosSpawnPlayer(Player, Now + 1000);
		}
	}
}

// ---------------------------------- //
// 2. Frenzy finale: last seconds of the round, everything turns golden
if (S_FrenzyFinale && !G_FrenzyActive && !G_InOvertime
	&& G_RoundEndTime - Now <= ML::Clamp(S_FrenzyDuration, 10, 300) * 1000) {
	G_FrenzyActive = True;
	declare K_ChaosCP[] GoldenCPs;
	foreach (CP in G_ActiveCPs) {
		declare Golden = CP;
		Golden.IsGolden = True;
		Golden.Points = C_PointsGolden;
		GoldenCPs.add(Golden);
	}
	G_ActiveCPs = GoldenCPs;
	RefreshMarkers();
	RefreshHud();
	Announce("$fd0FRENZY! Every checkpoint is golden!");
	Chat("$fd0FRENZY FINALE! $fffAll checkpoints are worth $fd0+" ^ C_PointsGolden ^ "$fff and an extra one is live!");
}

// ---------------------------------- //
// 3. Scheduled shuffle: unclaimed checkpoints teleport
if (G_NextShuffleTime > 0 && !G_InOvertime && Now >= G_NextShuffleTime) {
	G_NextShuffleTime = Now + ML::Clamp(S_ShuffleInterval, 15, 600) * 1000;
	declare NbToRespawn = G_ActiveCPs.count;
	G_ActiveCPs.clear();
	for (I, 1, NbToRespawn) {
		SpawnChaosCP(-1, False);
	}
	RefreshMarkers();
	Chat("$0cfShuffle! $fffAll checkpoints just teleported.");
}

// ---------------------------------- //
// 4. Proximity claims: first spawned player inside the radius takes it
declare K_ChaosCP[] RemainingCPs;
declare NbClaimed = 0;
foreach (CP in G_ActiveCPs) {
	declare CSmPlayer Claimer <=> Null;
	foreach (Player in Players) {
		if (Player.SpawnStatus != CSmPlayer::ESpawnStatus::Spawned) continue;
		if (ML::Distance(Player.Position, CP.Position) <= ML::Max(S_ClaimRadius, 4.)) {
			Claimer <=> Player;
			break;
		}
	}
	if (Claimer == Null) {
		RemainingCPs.add(CP);
		continue;
	}

	// Claimed!
	declare Login = Claimer.User.Login;
	declare Points = CP.Points;
	NbClaimed += 1;

	// Optional: first-claim bonus
	if (!G_FirstClaimDone) {
		G_FirstClaimDone = True;
		if (S_FirstClaimBonus > 0) {
			Points += ML::Max(S_FirstClaimBonus, 0);
			Chat("$0f0First blood! $<$fff" ^ Claimer.User.Name ^ "$> gets $0f0+" ^ S_FirstClaimBonus ^ "$fff bonus.");
		}
	}

	// Optional: combo chains
	if (S_ComboWindow > 0) {
		if (G_LastClaimTime.existskey(Login) && Now - G_LastClaimTime[Login] <= S_ComboWindow * 1000) {
			G_ComboCount[Login] = G_ComboCount[Login] + 1;
			declare ComboBonus = ML::Min(G_ComboCount[Login] - 1, C_ComboMaxBonus);
			Points += ComboBonus;
			Announce("$f0fx" ^ G_ComboCount[Login] ^ " COMBO! $<$fff" ^ Claimer.User.Name ^ "$> $f0f+" ^ ComboBonus ^ " bonus");
		} else {
			G_ComboCount[Login] = 1;
		}
		G_LastClaimTime[Login] = Now;
	}

	AddRoundPoints(Claimer.Score, Points);

	if (CP.IsGolden) {
		Announce("$fd0★ $<$fff" ^ Claimer.User.Name ^ "$> grabbed a GOLDEN checkpoint! $fd0+" ^ Points);
		Chat("$fd0★ $<$fff" ^ Claimer.User.Name ^ "$> claimed a golden checkpoint! $fd0+" ^ Points ^ "$fff (" ^ GetRoundPoints(Claimer.Score) ^ " pts)");
	} else {
		Chat("$0cf◆ $<$fff" ^ Claimer.User.Name ^ "$> claimed a checkpoint! $0cf+" ^ Points ^ "$fff (" ^ GetRoundPoints(Claimer.Score) ^ " pts)");
	}

	// Optional: golden checkpoints steal a point from the round leader
	if (CP.IsGolden && S_GoldenSteal) {
		declare CSmScore Victim <=> Null;
		declare VictimPoints = 0;
		foreach (Score in Scores) {
			if (Score == Claimer.Score) continue;
			if (GetRoundPoints(Score) > VictimPoints) {
				VictimPoints = GetRoundPoints(Score);
				Victim <=> Score;
			}
		}
		if (Victim != Null && VictimPoints > 0 && Victim.User != Null) {
			SetRoundPoints(Victim, VictimPoints - 1);
			Chat("$f00⚔ $<$fff" ^ Claimer.User.Name ^ "$> stole a point from $<$fff" ^ Victim.User.Name ^ "$>!");
		}
	}

	// Instant win condition
	if (S_PointsLimit > 0 && GetRoundPoints(Claimer.Score) >= S_PointsLimit) {
		G_RoundEndRequested = True;
	}
}
if (NbClaimed > 0) {
	G_ActiveCPs = RemainingCPs;

	// Any claim during overtime is sudden death
	if (G_InOvertime) {
		G_RoundEndRequested = True;
	}
}

// ---------------------------------- //
// 5. Keep the configured number of chaos checkpoints alive
//    (skipped in overtime: there is exactly one decider checkpoint)
if (!G_InOvertime && G_ActiveCPs.count < GetNbChaosCheckpoints()) {
	declare Missing = GetNbChaosCheckpoints() - G_ActiveCPs.count;
	for (I, 1, Missing) {
		SpawnChaosCP(-1, False);
	}
	RefreshMarkers();
}
if (NbClaimed > 0) {
	RefreshHud();
}

// ---------------------------------- //
// 6. End of round: points limit, timer, or overtime resolution
if (G_RoundEndRequested) {
	MB_StopRound();
} else if (!G_InOvertime && Now >= G_RoundEndTime) {
	if (S_Overtime && IsRoundTied()) {
		// Sudden death: one golden decider checkpoint, first claim wins
		G_InOvertime = True;
		G_FrenzyActive = False;
		G_OvertimeEndTime = Now + C_OvertimeMaxMs;
		G_ActiveCPs.clear();
		SpawnChaosCP(-1, True);
		RefreshMarkers();
		RefreshHud();
		SetHudEndTime(G_OvertimeEndTime);
		Announce("$f00OVERTIME! $fffFirst checkpoint wins the round!");
		Chat("$f00OVERTIME! $fffThe round is tied — first player to claim the golden checkpoint wins!");
	} else {
		MB_StopRound();
	}
} else if (G_InOvertime && Now >= G_OvertimeEndTime) {
	MB_StopRound();
}
***

***Match_EndRound***
***
SetHudEndTime(-1);
UIManager.UIAll.ClearMarkers();
G_ActiveCPs.clear();
foreach (Player in Players) {
	UnspawnPlayer(Player);
}

// Round winner = best round score
declare Winner <=> GetRoundLeader();
if (Winner != Null && Winner.User != Null && IsRoundTied()) {
	Announce("Round " ^ G_RoundNb ^ " ends in a draw!");
	Chat("$f80Round " ^ G_RoundNb ^ " over — it's a draw, everyone keeps their points.");
} else if (Winner != Null && Winner.User != Null) {
	Announce("$<$fff" ^ Winner.User.Name ^ "$> wins round " ^ G_RoundNb ^ " with $ff0" ^ GetRoundPoints(Winner) ^ "$fff points!");
	Chat("$0f0Round " ^ G_RoundNb ^ " over — $<$fff" ^ Winner.User.Name ^ "$> $0f0wins with $ff0" ^ GetRoundPoints(Winner) ^ "$0f0 points!");
} else {
	Announce("Round " ^ G_RoundNb ^ " over — nobody scored!");
	Chat("$f80Round " ^ G_RoundNb ^ " over — nobody scored a single point.");
}

TransferRoundPointsToMatch();
RefreshHud();
MB_Sleep(C_EndRoundPauseMs);

if (G_RoundNb >= ML::Max(S_RoundsPerMap, 1)) {
	MB_StopMap();
}
